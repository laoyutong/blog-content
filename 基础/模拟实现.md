## instanceof
```js
const myInstanceof = (a, b) => {
  if (!a || typeof a !== "object" && typeof a !== "function") {
    return false;
  }

  const target = b.prototype;
  let proto = Object.getPrototypeOf(a)

  while (proto) {
    if (proto === target) {
      return true;
    }
    proto = Object.getPrototypeOf(proto)
  }

  return false;
};
```

## new
```js
const myNew = (Construct, ...args) => {
  const obj = Object.create(Construct.prototype);
  const newObj = Construct.apply(obj, args);

  if (newObj && typeof newObj === "object") {
    return newObj;
  }

  return obj;
};
```

## Object.create
```js
const myCreate = (proto) => {
  function F() {}
  F.prototype = proto;

  const obj = new F();

  if (proto === null) {
    Object.setPrototypeOf(obj, null);
  }

  return obj;
};

```
## call
```js
const myCall = (context, ...args) => {
  context = context || window;
  const tempField = Symbol();
  context[tempField] = this;
  const result = context[tempField](...args)
  delete context[tempField]
  return result
};
```

## 防抖节流
```js
const throttle = (fn, delay) => {
  let flag = false;

  return (...args) => {
    if (flag) return;

    fn.call(this, ...args);
    flag = true;

    setTimeout(() => {
      flag = false;
    }, delay);
  };
};

const debounce = (fn, delay) => {
  let timer = null;
  
  return (...args) => {
    clearTimeout(timer);
    timer = setTimeout(() => {
      fn.call(this, ...args);
    }, delay);
  };
};

```
