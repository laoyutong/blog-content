## 异步并发控制

### 单方法处理

```js
const limitFn = (fn, limit) => {
  let count = 0;
  const queue = [];

  const run = () => {
    if (!queue.length || count >= limit) {
      return;
    }

    count++;

    const { args, resolve, reject } = queue.shift();

    fn(...args)
      .then(resolve)
      .catch(reject)
      .finally(() => {
        count--;
        run();
      });
  };

  return (...args) =>
    new Promise((resolve, reject) => {
      queue.push({ args, resolve, reject });
      run();
    });
};
```

### 批量版本

#### 工人池模式

```js
async function poolLimit(tasks, limit) {
  const result = [];
  const queue = tasks.map((fn, index) => ({ fn, index }));

  const coreWork = async () => {
    const { fn, index } = queue.shift();
    result[index] = await fn();

    if (queue.length) {
      await coreWork();
    }
  };

  await Promise.all(
    Array.from({ length: Math.min(limit, tasks.length) }, coreWork),
  );

  return result;
}
```

#### 递归版本

```js
async function poolLimit(tasks, limit) {
  const result = [];
  let index = 0;
  let running = 0;

  return new Promise((resolve) => {
    const run = () => {
      if (index === tasks.length && !running) {
        resolve(result);
      }

      while (running < limit && index < tasks.length) {
        running++;
        const i = index++;
        tasks[i]().then((res) => {
          result[i] = res;
          running--;
          run();
        });
      }
    };

    run();
  });
}
```

## 标签分类

### 单标签版本
```js
// const data = [
//   {
//     tag: "a",
//     value: 1,
//   },
//   {
//     tag: "b",
//     value: 2,
//   },
//   {
//     tag: "a",
//     value: 3,
//   },
// ];
const main = (origin) => {
  const result = {};
  origin.forEach((item) => {
    if (!result[item.tag]) {
      result[item.tag] = [];
    }
    result[item.tag].push(item.value);
  });
  return result
};
```

### 多标签版本
```js
// 输入数据：[
//   {
//     name: 1,
//     tags: ["a", "b",],
//   },
//   {
//     name: 2,
//     tags: ["a", "c"],
//   },                     
//   {
//     name: 3,
//     tags: ["a", "c"],
//   },
//   {
//     name: 4,
//     tags: ["b", "c"],
//   },
// ];

// 输出格式：
// [{
//   tag:'',
//   list:[],
//   children:[],
// }]
const main = (origin) => {
  const result = [];

  origin.forEach((originItem) => {
    let current = result;

    originItem.tags.forEach((tag, index) => {
      const isLastTag = index === originItem.tags.length - 1;
      const currentItem = current.find((i) => i.tag === tag);

      if (isLastTag) {
        if (currentItem) {
          currentItem.list.push(originItem);
        } else {
          current.push({
            tag,
            list: [originItem],
            children: [],
          });
        }
      } else {
        if (currentItem) {
          current = currentItem.children;
        } else {
          const newChildren = [];
          current.push({ tag, list: [], children: newChildren });
          current = newChildren;
        }
      }
    });
  });

  return result;
};
```
