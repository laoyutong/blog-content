### 快排
```js
function main(arr) {
  if (arr.length <= 1) {
    return arr;
  }

  const _partition = (left, right) => {
    const standard = arr[right];
    let activeIdx = left;

    for (let i = activeIdx; i < right; i++) {
      if (arr[i] < standard) {
        [arr[i], arr[activeIdx]] = [arr[activeIdx], arr[i]];
        activeIdx++;
      }
    }

    [arr[activeIdx], arr[right]] = [arr[right], arr[activeIdx]];

    return activeIdx
  };

  const _sort = (left, right) => {
    if (left < right) {
      const middleIdx = _partition(left, right);
      _sort(left, middleIdx - 1);
      _sort(middleIdx + 1, right);
    }
  };

  _sort(0, arr.length - 1);

  return arr;
}
```

## 卖股票
给定一个数组 prices ，它的第 i 个元素 prices[i] 表示一支给定股票第 i 天的价格
你只能选择 某一天 买入这只股票，并选择在 未来的某一个不同的日子 卖出该股票。设计一个算法来计算你所能获取的最大利润
返回你可以从这笔交易中获取的最大利润。如果你不能获取任何利润，返回 0 

```js
const main = (prices) => {
  let minPrice = Infinity;
  let maxPrice = 0;

  for (let i = 0; i < prices.length; i++) {
    const activePrice = prices[i];
    const money = activePrice - minPrice; 
    minPrice = Math.min(minPrice, activePrice);
    maxPrice = Math.max(maxPrice, money); 
  }

  return maxPrice;
};
```

## 盛水最多的容器
给定一个长度为 n 的整数数组 height 
有 n 条垂线，第 i 条线的两个端点是 (i, 0) 和 (i, height[i]) 
找出其中的两条线，使得它们与 x 轴共同构成的容器可以容纳最多的水
```js
const main = (input) => {
  let i = 0,
    j = input.length - 1;
  let max = 0;
  
  while (i < j) {
    const width = j - i;
    const height = Math.min(input[i], input[j]);
    max = Math.max(width * height, max);

    if(input[i]<input[j]){ // 短板决定容量，移动长板不可能更大
      i++
    }else{
      j--
    }
  }
  return max
};
```

```js
const main = (input) => {

    max = 0;
```

```js
function main(str) {

  for (const s of str) {
    if (s in map) {
      stack.push(s);
    } else {
```

T0 = 0, T1 = 1, T2 = 1, 且在 n >= 0 的条件下 Tn+3 = Tn + Tn+1 + Tn+2

```js
// 递归
function main(n, cache = {}) {
  if (n === 0 || n === 1) {
    return n;
  }
  
  if (n === 2) {
    return 1;
  }

  if (cache[n]) {
    return cache[n];
  }

  cache[n] = main(n - 1, cache) + main(n - 2, cache) + main(n - 3, cache);
  return cache[n];
}

// 动态规划
function main(n) {
```

## 不同路径
一个机器人位于一个 m x n 网格的左上角 ，每次只能向下或者向右移动一步
机器人试图达到网格的右下角，总共有多少条不同的路径
```js
// 普通版本
function main(m, n) {
  const array = new Array(m).fill(new Array(n).fill(0));

  for (let i = 0; i < m; i++) {
    array[i][0] = 1;
  }

  for (let j = 0; j < n; j++) {
    array[0][j] = 1;
  }

  for (let i = 1; i < m; i++) {
    for (let j = 1; j < n; j++) {
      array[i][j] = array[i - 1][j] + array[i][j - 1];
    }
  }

  return array[m - 1][n - 1];
}

// 滚动数组版本
function main(m, n) {

      f[j] += f[j - 1];
    }
  }
```

```js
function main(input) {

  let maxLength = 1;
  for (let i = 1; i < input.length; i++) {
    for (let j = 0; j < i; j++) {

        list[i] = Math.max(list[i], list[j] + 1);
```

## 数组中的第 K 个最大元素
给定整数数组 nums 和整数 k，请返回数组中第 k 个最大的元素
```js
function main(input, target) {
  const _helper = (left, right) => {
    if (left === right) {
      return input[left];
    }

    const cur = input[right];
    let index = left;
    for (let i = index; i < right; i++) {
      if (input[i] > cur) {
        if (i !== index) {
          [input[i], input[index]] = [input[index], input[i]];
        }
        index++;
      }
    }
    [input[index], input[right]] = [input[right], input[index]];

    if (index + 1 === target) {
      return input[index];
    }

    if (index < target) {
      return _helper(index + 1, right);
    } else {
      return _helper(left, index - 1);
    }
  };

  return _helper(0, input.length - 1);
}
```

```js
function main(s, t) {
  let i = 0;
  j = 0;

  while (i < s.length && j < t.length) {
    if (s[i] === t[j]) {
      i++;
    }
    j++;
  }
```

## 找零钱
给定一个金额和一组面额，计算出如何用最少数量的硬币凑齐该金额
```js
const main = (coins, amount) => {
  const result = new Array(amount + 1).fill(Infinity);
  result[0] = 0; // 凑齐 0 元需要 0 个硬币

  for (let i = 1; i <= amount; i++) {
    for (const coin of coins) {
      if (coin <= i) {
        result[i] = Math.min(result[i], result[i - coin] + 1); // 状态转移
      }
    }
  }

  return result[amount] === Infinity ? -1 : result[amount];
};
```
